\begin{frame}{Conceptualizing supply, requirements and distribution}
    \begin{columns}[T]
        \column{0.32\textwidth}
        \textbf{Supply}
        \begin{itemize}
            \item Stock-and-flow, not a registrant headcount
            \item Registered $\rightarrow$ active $\rightarrow$ practising in Ontario
            \item Converted to FTE (full/part-time, hours)
        \end{itemize}
        \column{0.32\textwidth}
        \textbf{Requirements}
        \begin{itemize}
            \item Needs-based: workforce required to meet projected population and service needs
            \item Driven by age/sex, population growth, care setting, models of care
            \item Not the same as current employment or vacancies
        \end{itemize}
        \column{0.32\textwidth}
        \textbf{Geography}
        \begin{itemize}
            \item Modelled by Ontario Health region
            \item Urban / rural / northern lens
            \item Province-wide balance can hide regional shortages
        \end{itemize}
    \end{columns}
    \vspace{0.4cm}
    \centering\textbf{Gap}$_{r,t}$ = Requirement$_{r,t}$ $-$ Supply$_{r,t}$ \quad (region $r$, year $t$, in FTEs)
\end{frame}

    \vspace{0.3cm}
    \begin{itemize}

        \item Moves from a \emph{descriptive forecast} to \emph{decision support}.
    \end{itemize}

\begin{frame}{Supply: stock-and-flow (cohort projection)}
    \[
    S_{r,a}(t+1) = S_{r,a}(t) + E_{r}(t) - A_{r,a}(t) + M_{r,a}(t)
    \]
    \vspace{-0.2cm}
    \begin{columns}[T]
        \column{0.55\textwidth}
        \begin{itemize}
            \item \textbf{Inflows}
            \begin{itemize}
                \item New graduates $E$, lagged by program length and exam pass rates; IEN licensure
                \item Net interprovincial / international migration $M$
                \item Re-entrants
            \end{itemize}
            \item \textbf{Outflows}
            \begin{itemize}
                \item Age- and career-stage-specific retirement / exit hazard rates $A$ (not one flat attrition rate)
            \end{itemize}
        \end{itemize}
        \column{0.42\textwidth}
        \begin{itemize}
            \item \textbf{Key assumptions}
            \begin{itemize}
                \item Age-specific rates stay at recent levels (baseline)
                \item Graduates and IENs stay in Ontario at observed rates
                \item FTE per active RN by age and sex
                \item Regional distribution of new entrants follows observed patterns
            \end{itemize}
        \end{itemize}
    \end{columns}
\end{frame}

\begin{frame}{Requirements: needs-based model}
    \[
    R_{r}(t) = \sum_{a,s,c} P_{r,a,s}(t) \times U_{a,s,c} \times I_{c}(t)
    \]
    \begin{itemize}
        \item $P$: population by region, age band and sex (Ontario Ministry of Finance projections)
        \item $U$: service utilization rate by age/sex and care setting $c$ (hospital, long-term care, home and community, primary care)
        \item $I$: service-intensity adjustment converting service use to RN FTEs, including policy-driven changes (more home care, skill-mix shifts)
        \item \textbf{Cross-check:} ratio benchmarking (RNs per 1,000 population) as a sensitivity lens on the needs-based estimate
        \item \textbf{Key assumptions:} utilization by age stays constant in the baseline; changes in care models enter only as explicit scenarios
    \end{itemize}
\end{frame}

\begin{frame}{Optimization layer: from forecast to recommendation}
    \textbf{Decision variables} $x_{t}$: domestic training seats, IEN licences fast-tracked, retention-incentive spend, inter-regional deployment incentives.
    \vspace{0.2cm}
    \begin{columns}[T]
        \column{0.48\textwidth}
        \textbf{Formulation A: least cost}
        \[
        \min \sum_{k,r,t} c_{k}\, x_{k,r,t}
        \]
        s.t. $S_{r,t}(x) \ge R_{r,t}$ for all $r,t$
        \column{0.48\textwidth}
        \textbf{Formulation B: fixed budget}
        \[
        \min \; \max_{r,t}\, \big(R_{r,t} - S_{r,t}(x)\big)^{+}
        \]
        s.t. $\sum c_{k}\, x_{k,r,t} \le B$
    \end{columns}
    \vspace{0.2cm}
    \begin{itemize}
        \item \textbf{Constraints:} training and clinical-placement capacity, annual budget ceilings, realistic ramp-up rates, minimum service level per region.
        \item Linear / mixed-integer program, open-source solvers (Python with PuLP or OR-Tools): fast and explainable.
        \item The two formulations are two views of the same cost--gap trade-off.
    \end{itemize}
\end{frame}

\begin{frame}{Optimal hiring policy model}
    \textbf{Decision variable} $x_{t}$: new nurse hired at time $t$.
    \vspace{0.2cm}
    \[
    \min_{x_{t}} \sum_{t=0}^{T} e^{-\lambda t}\, y_{t}^{+} + \sum_{t=0}^{T} e^{-\mu t}\, y_{t}^{-}
    \]
    s.t. 
    \begin{equation}
        d_{t} = \alpha_t\ p_{t}\quad \forall t
    \end{equation}

    \begin{equation}
        s_{t} = \beta_t\ n_{t}\quad \forall t
    \end{equation}

    \begin{equation}
        d_{t} - s_{t} = y_{t}^{+} - y_{t}^{-}  \quad \forall t
    \end{equation}

    \begin{equation}
        n_{t} = \gamma\ n_{t-1} + x_{t} \quad \forall t
    \end{equation}

    \begin{equation}
        x_{t}, y_{t}^{+}, y_{t}^{-} \ge 0 \quad \forall t
    \end{equation}
    
    \vspace{0.2cm}
    \begin{itemize}
        \item \textbf{Constraints:} training and clinical-placement capacity, annual budget ceilings, realistic ramp-up rates, minimum service level per region.
    \end{itemize}
\end{frame}

\begin{frame}{Data quality, gaps and privacy}
    \begin{columns}[T]
        \column{0.55\textwidth}
        \textbf{Quality and gaps}
        \begin{itemize}
            \item Registration overstates active, practising supply $\rightarrow$ adjust using employment status and FTE conversion
            \item Small regional cells are suppressed or unstable $\rightarrow$ pool years or regions
            \item 1--2 year publication lag $\rightarrow$ nowcast the latest year and flag it
            \item Self-reported hours are noisy $\rightarrow$ validate against multiple sources
            \item Reconcile definitions across CNO, CIHI and StatCan; document every assumption
        \end{itemize}
        \column{0.42\textwidth}
        \textbf{Privacy}
        \begin{itemize}
            \item Aggregate, de-identified data only
            \item No individual licensing or patient records needed
            \item Respect CIHI/CNO data-use terms and small-cell suppression rules
            \item Follow FIPPA principles for any internal administrative data
        \end{itemize}
    \end{columns}
\end{frame}

\section{Uncertainty}
\begin{frame}{Assessing uncertainty: scenarios and sensitivity}
    \begin{columns}[T]
        \column{0.48\textwidth}
        \textbf{Scenarios}
        \begin{itemize}
            \item \textbf{Baseline:} status-quo trends
            \item \textbf{High attrition:} accelerated retirements
            \item \textbf{Expanded seats:} policy intervention
            \item \textbf{High IEN:} fast-tracked licensing / immigration
        \end{itemize}
        \column{0.48\textwidth}
        \textbf{Sensitivity analysis}
        \begin{itemize}
            \item One-at-a-time variation of attrition, utilization growth, completion/pass rates and migration $\rightarrow$ tornado chart of which assumptions move the 2036 gap most
            \item Monte Carlo draws from plausible parameter ranges $\rightarrow$ uncertainty band around supply and requirements
        \end{itemize}
    \end{columns}
    \vspace{0.3cm}
    \begin{alertblock}{Robust recommendations}
        Re-solve the optimization under each scenario: show how the policy mix and its cost change, and favour options that perform well across scenarios.
    \end{alertblock}
\end{frame}

\section{Results \& Policy}
\begin{frame}{Communicating results to senior management}
    \begin{enumerate}
        \item \textbf{Headline chart:} projected supply vs.\ requirements, 2026--2036, province-wide and by region, with a shaded uncertainty band.
        \item \textbf{Regional gap view:} when and where gaps first appear.
        \item \textbf{Optimization result:} the lowest-cost policy mix (seats, IEN, retention, by year) as a stacked bar chart.
        \item \textbf{Tornado chart:} which assumptions matter most, and so where better data is worth the effort.
    \end{enumerate}
    \vspace{0.2cm}
    \begin{itemize}
        \item Plain language, one message per slide, assumptions stated up front.
        \item Brief for executives; technical appendix for analysts; reproducible code.
    \end{itemize}
\end{frame}

\begin{frame}{Limitations and policy implications}
    \begin{columns}[T]
        \column{0.48\textwidth}
        \textbf{Limitations}
        \begin{itemize}
            \item Quality depends on data lag and assumption validity
            \item Does not capture non-linear shocks (e.g.\ pandemics)
            \item Regional model is coarser than facility-level planning
            \item Requirements reflect current care models unless scenarios change them
        \end{itemize}
        \column{0.48\textwidth}
        \textbf{Policy options (with trade-offs)}
        \begin{itemize}
            \item Expand training seats: durable but slow
            \item Fast-track IEN licensing: quick, limited by pathway capacity
            \item Retention incentives: fast, recurring cost
            \item Regional / rural incentives: target maldistribution
            \item Scope-of-practice / skill-mix change: shifts requirements
        \end{itemize}
    \end{columns}
    \vspace{0.2cm}
    \centering Each option is presented with cost, lead time and feasibility to inform government action.
\end{frame}

\begin{frame}{Summary}
    \begin{itemize}
        \item A linked \textbf{supply} and \textbf{needs-based requirements} model, by region, over ten years.
        \item An \textbf{optimization layer} turns gaps into a least-cost policy mix.
        \item \textbf{Scenarios and sensitivity analysis} make recommendations robust.
        \item Built on public, aggregate data, with explicit quality and privacy safeguards.
        \item Communicated as options with trade-offs, refreshed annually as new data arrive.
    \end{itemize}
