\section{Demand-side Modelling}
\begin{frame}{Demand-side Modelling}
    \begin{itemize}
        \item Demand of healthcare services: $d_{t}$ [hours/time]
        \item $\alpha$: per capita demand for healthcare services [hours/(person time)]
        \item Population-driven demand:
        \begin{equation*}
            d_{t} = \alpha\ p_{t} \quad \forall t\quad\quad \text{[hours/time]}
        \end{equation*}
        \item Age-based demand: $d_{t} = \sum_{a} \alpha_{a}\ p_{a,t}$
        \item Specialty-based demand: $d_{t,s} = \sum_{a} \alpha_{a,s}\ p_{a,t}$
    \end{itemize}
\end{frame}

\begin{frame}{Population Modelling}
    \begin{itemize}
        \item Leslie (age-structured) model, in vector form:
        \begin{equation*}
            \mathbf{p}_{t+1} = \mathbf{L}\ \mathbf{p}_{t} + \mathbf{i}_{t} \quad \forall t
        \end{equation*}
        \item $\mathbf{p}_{t}$: population vector at time $t$.
        \item $\mathbf{L}$: Leslie matrix.
        \item $\mathbf{i}_{t}$: net immigration vector at time $t$ (major source of uncertainty).
    \end{itemize}

\end{frame}


\section{Supply-side Modelling}
\begin{frame}{Supply-side Modelling}
    \begin{itemize}
        \item Supply of healthcare services: $s_{t}$ [hours/time]
        \item $\beta_t$: per capita supply of healthcare services [hours/(FTE time)]
        \item Supply of healthcare services:
        \begin{equation*}
            s_{t} = \beta\ n_{t} \quad \forall t \quad\quad \text{[hours/time]}
        \end{equation*}
        \item Specialty-based supply: $s_{t,s} = \beta_{s}\ n_{s,t}$
    \end{itemize}
\end{frame}


\begin{frame}{Balance Surplus/Deficit}
    \begin{itemize}
        \item Balance of healthcare services:
        \begin{equation*}
            y_{t} = d_{t} - s_{t} \quad \forall t \quad\quad \text{[hours/time]}
        \end{equation*}
        \item Surplus: $y_{t}^{+} = \max(y_{t}, 0)$
        \item Deficit: $y_{t}^{-} = \max(-y_{t}, 0)$
    \end{itemize}
\end{frame}


\section{Optimal hiring policy model}
\begin{frame}{Optimal hiring policy model}
    \textbf{Decision variable} $x_{t}$: new RNs hired at time $t$.
    \vspace{0.2cm}
    \[
    \min_{x_{t}} \sum_{t=0}^{T} \lambda\ e^{-\delta t}\, y_{t}^{+} + \sum_{t=0}^{T} \mu\ e^{-\delta t}\, y_{t}^{-}
    \]
    s.t. 
    \begin{equation}
        d_{t} = \alpha_t\ p_{t}\quad \forall t
    \end{equation}

    \begin{equation}
        s_{t} = \beta_t\ n_{t}\quad \forall t
    \end{equation}

    \begin{equation}
        d_{t} - s_{t} = y_{t}^{+} - y_{t}^{-}  \quad \forall t
    \end{equation}

    \begin{equation}
        n_{t} = \gamma\ n_{t-1} + x_{t} \quad \forall t
    \end{equation}

    \begin{equation}
        x_{t}, y_{t}^{+}, y_{t}^{-} \ge 0 \quad \forall t
    \end{equation}
    
\end{frame}

\begin{frame}{Data quality, gaps and privacy}
    \begin{columns}[T]
        \column{0.55\textwidth}
        \textbf{Quality and gaps}
        \begin{itemize}
            \item Registration overstates active, practising supply $\rightarrow$ adjust using employment status and FTE conversion
            \item Small regional cells are suppressed or unstable $\rightarrow$ pool years or regions
            \item 1--2 year publication lag $\rightarrow$ nowcast the latest year and flag it
            \item Self-reported hours are noisy $\rightarrow$ validate against multiple sources
            \item Reconcile definitions across CNO, CIHI and StatCan; document every assumption
        \end{itemize}
        \column{0.42\textwidth}
        \textbf{Privacy}
        \begin{itemize}
            \item Aggregate, de-identified data only
            \item No individual licensing or patient records needed
            \item Respect CIHI/CNO data-use terms and small-cell suppression rules
            \item Follow FIPPA principles for any internal administrative data
        \end{itemize}
    \end{columns}
\end{frame}

    \end{enumerate}
    \vspace{0.2cm}
    \begin{itemize}
        \item Plain language, one message per slide, assumptions stated up front.
        \item Brief for executives; technical appendix for analysts; reproducible code.
    \end{itemize}

\begin{frame}{Limitations and policy implications}
    \begin{columns}[T]
        \column{0.48\textwidth}
        \textbf{Limitations}
        \begin{itemize}
            \item Quality depends on data lag and assumption validity
            \item Does not capture non-linear shocks (e.g.\ pandemics)
            \item Regional model is coarser than facility-level planning
            \item Requirements reflect current care models unless scenarios change them
        \end{itemize}
        \column{0.48\textwidth}
        \textbf{Policy options (with trade-offs)}
        \begin{itemize}
            \item Expand training seats: durable but slow
            \item Fast-track IEN licensing: quick, limited by pathway capacity
            \item Retention incentives: fast, recurring cost
            \item Regional / rural incentives: target maldistribution
            \item Scope-of-practice / skill-mix change: shifts requirements
        \end{itemize}
    \end{columns}
    \vspace{0.2cm}
    \centering Each option is presented with cost, lead time and feasibility to inform government action.
\end{frame}
